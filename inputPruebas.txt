function testComparaciones(): void {
    let c: char;
    let i: int;
    let n: number;

    c = 'B';
    i = 66;
    n = 66.5;

    log 'C', 'O', 'M', 'P', 'A', 'R', 'A', 'C', 'I', 'O', 'N', ':', '\n';

    if (c == i) { log '1', '.', ' ', 'E', 'Q', ' ', 'O', 'K', '\n'; }
    if (i < n)  { log '2', '.', ' ', 'L', 'T', ' ', 'O', 'K', '\n'; }
    if (n > c)  { log '3', '.', ' ', 'G', 'T', ' ', 'O', 'K', '\n'; }
}

function testAnidados(): void {
    let i, j, sum: int;

    log 'A', 'N', 'I', 'D', 'A', 'D', 'O', 'S', ':', '\n';

    i = 10;
    if (i > 5) {
        if (i == 10) {
            log 'I', 'f', ' ', 'O', 'K', '\n';
        } else {
            log 'E', 'R', 'R', 'O', 'R', '\n';
        }
    } else {
        log 'E', 'R', 'R', 'O', 'R', '\n';
    }

    sum = 0;
    i = 0;
    while (i < 3) {
        j = 0;
        while (j < 2) {
            sum = sum + 1;
            j = j + 1;
        }
        i = i + 1;
    }

    if (sum == 6) {
        log 'W', 'h', 'i', 'l', 'e', ' ', 'O', 'K', '\n';
    } else {
        log 'E', 'R', 'R', 'O', 'R', '\n';
    }
}
